Section A: LIVE network attempt with the pipeline's real default fetcher

           real merger, real publisher. Live tvly was DNS-failing during the
           verification window, which is why transport is replayed — the
           change-detection logic itself (sha256 compare against persistent
           state) is fully real.

Everything runs inside /tmp/kb-evidence sandboxes. The repository's state/
and knowledge/ are never touched.
"""

                    format="%(levelname)s %(name)s: %(message)s")

# Real internet content fetched by this pipeline earlier today (production run):
REAL_MD = (REPO / "state" / "cache" /
           "7ed29293dfd8a66fc7f9330001c9e02a43ef6447cfbbbbb03d703a6fec4a8337.md"
           ).read_text(encoding="utf-8")


def sha(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def replay_fetcher(content_by_url: dict):
    """Same injection point tests/test_orchestrate.py uses (fetcher=...)."""
    def _fetch(urls, fetched_at):
        out = []
        for u in urls:
            content = content_by_url[u]
            out.append(FetchResult(
                url=u, strategy="replay-real-content", content_type="markdown",
                title=None, content=content, sha256=sha(content),
                fetched_at=fetched_at).to_json() | {"ok": True})
        return out
    return _fetch


def snapshot(kdir: Path) -> dict:
    kdir = Path(kdir)
    if not kdir.exists():
        return {}
    return {p.name: hashlib.sha256(p.read_bytes()).hexdigest()
            for p in sorted(kdir.glob("*.md"))}


def doc_table(kdir: Path) -> list:
    rows = []
    for p in sorted(Path(kdir).glob("*.md")):
        if p.name in ("INDEX.md", "CHANGELOG.md"):
            continue
        m = re.search(r"^id:\s*(\S+)", p.read_text(encoding="utf-8"), re.M)
        rows.append({"file": p.name, "id": m.group(1) if m else None})
    return rows


def section(title: str) -> None:
    print("\n" + "=" * 72 + "\n" + title + "\n" + "=" * 72, flush=True)


def fresh(name: str) -> dict:
    base = ROOT / name
    if base.exists():
        shutil.rmtree(base)
    return {"state": base / "fetch_state.json", "cache": base / "cache",
            "claims": base / "claims", "knowledge": base / "knowledge"}

}
print(json.dumps(verdicts, indent=2))
